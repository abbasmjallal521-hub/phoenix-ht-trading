const express = require('express'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; // قراءة الملفات من المجلد الرئيسي (Root) حيث يوجد index.html app.use(express.static(path.join(__dirname))); app.get('/health', (req, res) => { res.status(200).send('OK'); }); app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
